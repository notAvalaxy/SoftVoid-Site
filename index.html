// --- OCR & EXP Tracker Engine (Tuned for MapleStory UI) --- let video = document.getElementById('game-video'); let canvas = document.getElementById('ocr-canvas'); let ctx = canvas.getContext('2d'); let startBtn = document.getElementById('start-btn'); let isTracking = false; // Bounding box tailored for the bottom-right EXP area in the game UI // (Expressed as percentages of the total video resolution so it scales automatically) let roiPct = { x: 0.52, y: 0.90, width: 0.16, height: 0.08 }; // Chart.js Configuration const chartCtx = document.getElementById('expChart').getContext('2d'); const expChart = new Chart(chartCtx, { type: 'line', data: { labels: [], datasets: [{ label: 'EXP Percentage Trajectory', data: [], borderColor: '#93c5fd', backgroundColor: 'rgba(147, 197, 253, 0.1)', fill: true, tension: 0.25, pointRadius: 3 }] }, options: { responsive: true, plugins: { legend: { labels: { color: '#e0e0ff', font: { family: 'Quicksand' } } } }, scales: { x: { ticks: { color: '#a0a0c0' }, grid: { color: 'rgba(255,255,255,0.05)' } }, y: { ticks: { color: '#a0a0c0' }, grid: { color: 'rgba(255,255,255,0.05)' } } } } }); startBtn.addEventListener('click', async () => { try { const stream = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: false }); video.srcObject = stream; video.play(); isTracking = true; requestAnimationFrame(processFrame); setInterval(runOCR, 3000); // Scan every 3 seconds startBtn.textContent = "Tracking Active"; startBtn.style.background = "#059669"; } catch (err) { console.error("Screen capture cancelled or error:", err); } }); function processFrame() { if (!isTracking) return; if (video.videoWidth > 0) { canvas.width = video.videoWidth; canvas.height = video.videoHeight; ctx.drawImage(video, 0, 0, canvas.width, canvas.height); // Calculate pixel coordinates from percentages let rx = canvas.width * roiPct.x; let ry = canvas.height * roiPct.y; let rw = canvas.width * roiPct.width; let rh = canvas.height * roiPct.height; // Render the Region of Interest (ROI) blue outline box on the preview feed ctx.strokeStyle = '#3b82f6'; ctx.lineWidth = 3; ctx.strokeRect(rx, ry, rw, rh); } requestAnimationFrame(processFrame); } async function runOCR() { if (!isTracking || canvas.width === 0) return; let rx = canvas.width * roiPct.x; let ry = canvas.height * roiPct.y; let rw = canvas.width * roiPct.width; let rh = canvas.height * roiPct.height; // Crop the exact EXP area and scale it up 2x for cleaner OCR accuracy const cropCanvas = document.createElement('canvas'); cropCanvas.width = rw * 2; cropCanvas.height = rh * 2; const cropCtx = cropCanvas.getContext('2d'); cropCtx.imageSmoothingEnabled = false; // Keep pixel text sharp cropCtx.drawImage(canvas, rx, ry, rw, rh, 0, 0, cropCanvas.width, cropCanvas.height); try { // Pass whitelist characters to Tesseract so it focuses heavily on numbers and decimal/percentage text const result = await Tesseract.recognize(cropCanvas, 'eng', { tessedit_char_whitelist: '0123456789.%k' }); const rawText = result.data.text.trim(); console.log("OCR Detected:", rawText); // Extract numbers (handling decimals like 21.95) const match = rawText.match(/(\d+\.\d+|\d+)/); if (match) { const currentExp = parseFloat(match[0]); document.getElementById('current-exp').innerText = currentExp + "%"; const now = new Date().toLocaleTimeString(); expChart.data.labels.push(now); expChart.data.datasets[0].data.push(currentExp); if (expChart.data.labels.length > 25) { expChart.data.labels.shift(); expChart.data.datasets[0].data.shift(); } expChart.update(); } } catch (ocrErr) { console.warn("OCR scan iteration skipped:", ocrErr); } }
